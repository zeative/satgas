      tag: el.tagName.toLowerCase(),
      id: el.id || undefined,
      classes: typeof el.className === "string" ? el.className.slice(0, 400) : undefined,
      text: (el.innerText || "").trim().slice(0, 160),
      ownText: own && el.children.length === 0,
      html: el.outerHTML.slice(0, 900),
      selector: selector(el),
      loc: loc ? loc.getAttribute("data-satgas-loc") || loc.getAttribute("data-inspector-relative-path") : undefined,
      path: location.pathname,
      styles,
    };
  };
  const target = (e) => {
    const el = e.target;
    if (!el || el.nodeType !== 1 || el.closest("[data-satgas-ui]")) return null;
    return el === document.documentElement ? document.body : el;
  };
  const select = (el) => {
    // Picking the element already picked keeps its changes, on the page and in the panel alike.
    const same = picked === el;
    if (picked && !same) clear();
    picked = el;
    box = box || outline(false);
    redraw();
    post({ type: "pick", el: describe(el), keep: same });
  };
  const clear = () => {
    if (picked) for (const k of Object.keys(edits)) picked.style[k] = "";
    edits = {};
  };
  const FONTS = new Set();
  const loadFont = (family) => {
    const name = String(family).split(",")[0].replace(/["']/g, "").trim();
    if (!name || FONTS.has(name) || /^(system-ui|sans-serif|serif|monospace|ui-)/.test(name)) return;
    FONTS.add(name);
    const l = document.createElement("link");
    l.rel = "stylesheet";
    l.setAttribute("data-satgas-ui", "");
    l.href = `https://fonts.googleapis.com/css2?family=${encodeURIComponent(name).replace(/%20/g, "+")}:wght@300;400;500;600;700;800&display=swap`;
    document.head.appendChild(l);
  };

  addEventListener(
    "pointermove",
    (e) => {
      if (!picking) return;
      alt = e.altKey;
      hovered = target(e);
      hoverBox = hoverBox || outline(true);
      redraw();
    },
    true,
  );
  // Double-clicking a picked element that holds only its own words edits them in place, as in a design tool; the panel
  // follows along, and a click elsewhere, Enter or Escape ends it.
  let editing = null;
  const ownWords = (el) => !el.children.length && [...el.childNodes].some((c) => c.nodeType === 3 && c.textContent.trim());
  const stopEditing = () => {
    if (!editing) return;
    editing.removeAttribute("contenteditable");
    editing.removeEventListener("input", onEdit);
    editing = null;
    getSelection()?.removeAllRanges();
  };
  const onEdit = () => {
    if (!editing) return;
    redraw();
    post({ type: "text-edit", value: editing.textContent });
  };
  const startEditing = (el) => {
    editing = el;
    el.setAttribute("contenteditable", "plaintext-only");
    el.addEventListener("input", onEdit);
    el.focus();
    const range = document.createRange();
    range.selectNodeContents(el);
    getSelection()?.removeAllRanges();
    getSelection()?.addRange(range);
  };
  for (const type of ["click", "pointerdown", "mousedown", "pointerup", "mouseup", "dblclick", "submit"]) {
    addEventListener(
      type,
      (e) => {
        if (!picking) return;
        // Inside the words being edited the pointer places the caret and selects, as in any text field.
        if (editing && e.target instanceof Node && editing.contains(e.target)) return;
        e.preventDefault();
        e.stopPropagation();
        if (type === "dblclick") {
          const el = target(e);
          if (el && el === picked && ownWords(el)) startEditing(el);
        } else if (type === "click") {
          stopEditing();
          const el = target(e);
          if (el) select(el);
          // The cancelled mousedown kept focus out of the page; take it, so the layer keys reach this page.
          window.focus();
        }
      },
      true,
    );
  }
  // Moving through layers the way design tools do: Enter goes into the first child, Shift+Enter out to the parent, Tab
  // and Shift+Tab to the next and previous sibling.
  const walkLayers = (e) => {
    let next = null;
    if (e.key === "Enter") next = e.shiftKey ? picked.parentElement : kids(picked)[0];
    else if (e.key === "Tab") {
      const sibs = picked.parentElement ? kids(picked.parentElement) : [];
      const i = sibs.indexOf(picked);
      next = sibs[(i + (e.shiftKey ? -1 : 1) + sibs.length) % sibs.length];
    }
    if (!next || next === picked || next === document.documentElement) return;
    e.preventDefault();
    select(next);
  };
  addEventListener(
    "keydown",
    (e) => {
      if (picked && !editing && (e.metaKey || e.ctrlKey) && (e.key === "z" || e.key === "Z" || e.key === "y")) {
        e.preventDefault();
        post({ type: "undo", redo: e.shiftKey || e.key === "y" });
      } else if (editing && (e.key === "Escape" || (e.key === "Enter" && !e.shiftKey))) {
        e.preventDefault();
        stopEditing();
      } else if (e.key === "Escape" && (picking || picked)) post({ type: "escape" });
      else if (picking && picked && !e.target.closest?.("input,textarea,select,[contenteditable]")) walkLayers(e);
    },
    true,
  );

  addEventListener("message", (e) => {
    const m = e.data;
    if (e.source !== window.parent || !m || m.satgas !== true) return;
    if (m.type === "run") {
      if (m.mode === "live" || m.mode === "idle" || m.mode === "still") {
        const was = mode;
        mode = m.mode;
        // Coming back from a pan into idle waits the full quiet time again rather than staying frozen.
        if (mode !== was) settle();
      }
    } else if (m.type === "screen") {
      screenH = Number(m.h) || 0;
      document.documentElement.style.overflowY = screenH ? "hidden" : "";
      lastH = 0;
      size();
    } else if (m.type === "mode") {
      picking = !!m.pick;
      if (!picking) stopEditing();
      document.documentElement.style.cursor = picking ? "crosshair" : "";
      if (!picking) hovered = null;
      redraw();
    } else if (m.type === "style" && picked) {
      for (const [k, v] of Object.entries(m.css || {})) {
        if (k === "fontFamily") loadFont(v);
        picked.style[k] = v;
        edits[k] = v;
      }
      redraw();
      post({ type: "pick", el: describe(picked), keep: true });
    } else if (m.type === "select") {
      const el = refs[Number(m.ref)];
      if (el?.isConnected) select(el);
    } else if (m.type === "hover") {
      const el = m.ref === null || m.ref === undefined ? null : refs[Number(m.ref)];
      hovered = el?.isConnected ? el : null;
      hoverBox = hoverBox || outline(true);
      place(hoverBox, hovered && hovered !== picked ? hovered : null);

    } else if (m.type === "attr" && picked) {
      const el = m.name === "href" ? picked.closest("a") : picked;
      if (el && /^[a-z-]+$/.test(String(m.name))) el.setAttribute(String(m.name), String(m.value));
    } else if (m.type === "tokens") {
      post({ type: "tokens", tokens: tokens() });
    } else if (m.type === "tokens-set") {
      for (const [k, v] of Object.entries(m.vars || {})) {
        if (!k.startsWith("--")) continue;
        document.documentElement.style.setProperty(k, String(v));
        tweaked.add(k);
      }
      redraw();
    } else if (m.type === "tokens-reset") {
      for (const k of tweaked) document.documentElement.style.removeProperty(k);
      tweaked.clear();
      redraw();
    } else if (m.type === "text" && picked) {
      if (picked === editing && picked.textContent === String(m.value)) return;
      picked.textContent = String(m.value);
      redraw();
    } else if (m.type === "restyle" && picked) {
      // The panel's undo and redo: the picked element goes back to exactly these changes.
      clear();
      for (const [k, v] of Object.entries(m.css || {})) {
        if (k === "fontFamily") loadFont(v);
        picked.style[k] = v;
        edits[k] = v;
      }
      redraw();
      post({ type: "pick", el: describe(picked), keep: true });
    } else if (m.type === "reset") {
      clear();
      redraw();
      if (picked) post({ type: "pick", el: describe(picked), keep: true });
    } else if (m.type === "deselect") {
      stopEditing();
      picked = null;
      edits = {};
      redraw();
    } else if (m.type === "go") {
      location.assign(String(m.path));
    } else if (m.type === "back") {
      history.back();
    } else if (m.type === "forward") {
      history.forward();
    }
  });
})();
