import type { ComponentType, ReactNode } from "react";
import { Link } from "react-router";
import { cn } from "@/lib/cn";

// The pieces every page shares. Their look comes from the tokens in index.css; a page picks a variant, never a colour.

const BUTTON = {
  primary: "bg-accent text-accent-foreground hover:bg-accent/90",
  secondary: "bg-surface text-foreground ring-1 ring-border hover:bg-muted",
  ghost: "text-foreground hover:bg-muted",
  inverse: "bg-background text-foreground hover:bg-background/90",
} as const;
const BUTTON_SIZE = { sm: "h-9 px-3.5 text-sm gap-1.5", md: "h-11 px-5 text-sm gap-2", lg: "h-12 px-6 text-base gap-2" } as const;

export function Button({
  variant = "primary",
  size = "md",
  to,
  href,
  icon: Icon,
  iconEnd: IconEnd,
  full = false,
  type = "button",
  disabled,
  onClick,
  children,
}: {
  variant?: keyof typeof BUTTON;
  size?: keyof typeof BUTTON_SIZE;
  to?: string;
  href?: string;
  icon?: ComponentType<{ className?: string }>;
  iconEnd?: ComponentType<{ className?: string }>;
  full?: boolean;
  type?: "button" | "submit";
  disabled?: boolean;
  onClick?: () => void;
  children: ReactNode;
}) {
  const cls = cn(
    "inline-flex items-center justify-center rounded-control font-medium whitespace-nowrap transition-[background-color,box-shadow,transform] duration-150 ease-out active:scale-[0.98] focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-accent disabled:pointer-events-none disabled:opacity-50",
    BUTTON[variant],
    BUTTON_SIZE[size],
    full && "w-full",
  );
  const body = (
    <>
      {Icon && <Icon className="size-4 shrink-0" />}
      {children}
      {IconEnd && <IconEnd className="size-4 shrink-0" />}
    </>
  );
  if (to) return <Link to={to} className={cls}>{body}</Link>;
  if (href) return <a href={href} className={cls}>{body}</a>;
  return <button type={type} disabled={disabled} onClick={onClick} className={cls}>{body}</button>;
}

const BADGE = {
  neutral: "bg-muted text-foreground",
  accent: "bg-accent/12 text-accent",
  success: "bg-success/12 text-success",
  warning: "bg-warning/15 text-warning",
  danger: "bg-danger/12 text-danger",
} as const;

export function Badge({ tone = "neutral", children }: { tone?: keyof typeof BADGE; children: ReactNode }) {
  return <span className={cn("inline-flex h-6 items-center rounded-full px-2.5 text-xs font-medium", BADGE[tone])}>{children}</span>;
}

const PAD = { sm: "p-4", md: "p-6", lg: "p-8" } as const;

export function Card({
  padding = "md",
  tone = "default",
  interactive = false,
  to,
  children,
}: {
  padding?: keyof typeof PAD;
  tone?: "default" | "muted" | "inverse";
  interactive?: boolean;
  to?: string;
  children: ReactNode;
}) {
  const cls = cn(
    "block rounded-card",
    { default: "bg-surface ring-1 ring-border shadow-card", muted: "bg-muted", inverse: "bg-foreground text-background" }[tone],
    PAD[padding],
    (interactive || to) && "transition-[box-shadow,transform] duration-200 ease-out hover:-translate-y-0.5 hover:shadow-lift",
  );
  return to ? <Link to={to} className={cls}>{children}</Link> : <div className={cls}>{children}</div>;
}

const RATIO = { square: "aspect-square", portrait: "aspect-[4/5]", landscape: "aspect-[4/3]", wide: "aspect-[16/9]", hero: "aspect-[16/10]" } as const;

// A real photo at a fixed ratio, cropped the same way everywhere. Never a pattern or a drawing in its place.
export function Photo({ src, alt, ratio = "landscape", radius = "card", priority = false }: { src: string; alt: string; ratio?: keyof typeof RATIO; radius?: "none" | "control" | "card" | "panel"; priority?: boolean }) {
  return (
    <div className={cn("overflow-hidden bg-muted", RATIO[ratio], { none: "", control: "rounded-control", card: "rounded-card", panel: "rounded-panel" }[radius])}>
      <img src={src} alt={alt} loading={priority ? "eager" : "lazy"} decoding="async" className="size-full object-cover" />
    </div>
  );
}

export function Skeleton({ lines = 3, block = false }: { lines?: number; block?: boolean }) {
  if (block) return <div className="h-40 animate-pulse rounded-card bg-muted motion-reduce:animate-none" />;
  return (
    <div className="flex flex-col gap-2">
      {Array.from({ length: lines }, (_, i) => (
        <div key={i} className={cn("h-3.5 animate-pulse rounded bg-muted motion-reduce:animate-none", i === lines - 1 ? "w-2/3" : "w-full")} />
      ))}
    </div>
  );
}

export function EmptyState({ icon: Icon, title, body, action }: { icon: ComponentType<{ className?: string }>; title: string; body: string; action?: ReactNode }) {
  return (
    <div className="flex flex-col items-center gap-3 px-6 py-14 text-center">
      <span className="flex size-11 items-center justify-center rounded-full bg-muted text-muted-foreground">
        <Icon className="size-5" />
      </span>
      <p className="text-base font-semibold text-foreground">{title}</p>
      <p className="max-w-sm text-sm text-muted-foreground">{body}</p>
      {action}
    </div>
  );
}
