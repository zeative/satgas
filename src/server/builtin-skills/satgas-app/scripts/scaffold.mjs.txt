// Satgas: tags every HTML element with the file and line that draws it, so the canvas can show which section a
// teammate is working on and the inspector can name its source. Only while developing; a build stays clean.
function satgasLoc() {
  // biome-ignore lint/suspicious/noExplicitAny: a Babel plugin, typed loosely to need no extra packages.
  return ({ types: t }: { types: any }) => ({
    visitor: {
      // biome-ignore lint/suspicious/noExplicitAny: see above.
      JSXOpeningElement(path: any, state: any) {
        const name = path.node.name;
        if (name.type !== "JSXIdentifier" || !/^[a-z]/.test(name.name)) return;
        // biome-ignore lint/suspicious/noExplicitAny: see above.
        if (path.node.attributes.some((a: any) => a.type === "JSXAttribute" && a.name?.name === "data-satgas-loc")) return;
        const file = relative(process.cwd(), state.filename ?? "").split("\\\\").join("/");
        const at = \`\${file}:\${path.node.loc?.start.line ?? 0}\`;
        path.node.attributes.push(t.jsxAttribute(t.jsxIdentifier("data-satgas-loc"), t.stringLiteral(at)));
      },
    },
  });
}

const base = process.env.SATGAS_APP_BASE || "/";

// The app's API (server/index.ts) starts and stops with the dev server, so one app_start runs both.
function satgasApi(): Plugin {
  return {
    name: "satgas-api",
    configureServer(server) {
      const child = spawn("bun", ["--watch", "server/index.ts"], { stdio: "inherit", env: { ...process.env, API_PORT: String(apiPort) } });
      const stop = () => child.kill();
      server.httpServer?.on("close", stop);
      process.on("exit", stop);
    },
  };
}
`
