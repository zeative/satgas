#!/usr/bin/env node
// Finds what makes a page look improvised or AI-made, in the source, without a model: colours and sizes outside the
// design tokens, styles that skip the kit, and the known tells. Exit code 1 when it finds any.
//   node ui-check.mjs [project dir]
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join, relative, resolve } from "node:path";

const root = resolve(process.argv[2] ?? ".");
const src = join(root, "src");
const files = [];
const walk = (dir) => {
  for (const name of readdirSync(dir)) {
    const full = join(dir, name);
    if (statSync(full).isDirectory()) {
      if (name !== "kit" && name !== "node_modules") walk(full);
    } else if (/\.(tsx|jsx)$/.test(name)) files.push(full);
  }
};
try {
  walk(src);
} catch {
  console.error(`No src folder in ${root}.`);
  process.exit(1);
}

const PALETTE = "slate|gray|zinc|neutral|stone|red|orange|amber|yellow|lime|green|emerald|teal|cyan|sky|blue|indigo|violet|purple|fuchsia|pink|rose";
const RULES = [
  ["raw colour", /(?<![\w-])#[0-9a-fA-F]{3,8}\b|\brgba?\(|\bhsla?\(|\boklch\(/, "use a token (bg-accent, text-muted-foreground…) from src/index.css"],
  ["Tailwind palette colour", new RegExp(`\\b(?:bg|text|border|ring|from|via|to|fill|stroke|outline|divide|shadow)-(?:${PALETTE})-\\d{2,3}\\b`), "use a token, so the page follows the design system"],
  ["arbitrary value", /\b[a-z][\w-]*-\[(?!--)[^\]]+\]/, "use a step from the scale; add a token if the system lacks one"],
  ["inline style", /\bstyle=\{\{/, "style through the kit or tokens"],
  ["gradient text", /\bbg-clip-text\b/, "a known AI tell; set headlines in a solid colour"],
  ["looping animation", /\banimate-(?:pulse|bounce|ping|spin)\b/, "only for a real loading state; use the kit's Skeleton"],
  ["tracked caps label", /\buppercase\b[^"'`]*\btracking-(?:wider|widest)\b|\btracking-(?:wider|widest)\b[^"'`]*\buppercase\b/, "a known AI tell (eyebrow labels); use sentence case"],
  ["italic accent word", /<(?:em|i|span)[^>]*\b(?:italic|font-serif)\b[^>]*>/, "a known AI tell (one italic serif word in a headline); keep it only if the target has it"],
  ["emoji", /[\u{1F300}-\u{1FAFF}\u{2600}-\u{27BF}]/u, "use an icon from lucide-react"],
];

// What the chosen target itself does is a choice, not a tell: DESIGN.md can allow rules by name, as
// "ui-check allows: tracked caps label, italic accent word".
const allowed = new Set(
  (/^ui-check allows?:\s*(.+)$/im.exec((() => { try { return readFileSync(join(root, "DESIGN.md"), "utf8"); } catch { return ""; } })())?.[1] ?? "")
    .split(",")
    .map((r) => r.trim().toLowerCase())
    .filter((r) => r && r !== "default font"),
);
const found = [];
for (const file of files) {
  const text = readFileSync(file, "utf8");
  const lines = text.split("\n");
  lines.forEach((line, i) => {
    for (const [name, re, fix] of RULES) if (!allowed.has(name) && re.test(line)) found.push({ file, line: i + 1, name, fix, code: line.trim().slice(0, 110) });
  });
  // A drawing made of many paths is a hand-made illustration, which always looks homemade next to real pictures.
  for (const m of text.matchAll(/<svg[\s\S]*?<\/svg>/g)) {
    if ((m[0].match(/<(?:path|circle|rect|ellipse|polygon)\b/g) ?? []).length > 4)
      found.push({ file, line: text.slice(0, m.index).split("\n").length, name: "hand-drawn illustration", fix: "use a real photo, the product's interface, or a lucide icon", code: "<svg … />" });
  }
}

// Type: a page in a default or stock face reads as generated before anything else on it is seen.
const NEVER_FONTS = /^(geist|geist mono|inter|inter tight|roboto|open sans|lato|poppins|montserrat|dm sans|manrope|plus jakarta sans|outfit|figtree|nunito|space grotesk|jost|playfair display|instrument serif|fraunces|cormorant|cormorant garamond|dm serif display|eb garamond|lora|libre baskerville|newsreader|cinzel)$/i;
const read = (path) => {
  try {
    return readFileSync(join(root, path), "utf8");
  } catch {
    return "";
  }
};
const cssPath = "src/index.css";
const css = read(cssPath);
const tokenOf = (name) => new RegExp(`--font-${name}\\s*:\\s*([^;]+);`).exec(css)?.[1]?.trim() ?? "";
const firstFamily = (value) => value.split(",")[0].trim().replace(/^["']|["']$/g, "");
const fontFinding = (fix, code) => found.push({ file: join(root, cssPath), line: 1, name: "default font", fix, code });
if (css) {
  const display = tokenOf("display");
  const text = tokenOf("sans");
  const used = [...new Set([display, text, tokenOf("mono")].map(firstFamily).filter((f) => f && !f.startsWith("var(")))];
  for (const family of used.filter((f) => NEVER_FONTS.test(f))) fontFinding(`${family} is on the never list; choose by character from satgas-design's fonts.md`, family);
  if (!display || display.startsWith("var(") || firstFamily(display) === firstFamily(text))
    fontFinding("headlines need a display face of their own (--font-display), different from the text face", display || "--font-display missing");
  for (const family of used.filter((f) => /^(ui-|system-ui|sans-serif|serif|monospace)/.test(f))) fontFinding("a system font is no choice; set a real family", family);
  const loaded = `${read("index.html")}\n${css}`;
  for (const family of used.filter((f) => !/^(ui-|system-ui|sans-serif|serif|monospace)/.test(f))) {
    const google = family.replace(/\s+/g, "+");
    const share = family.toLowerCase().replace(/\s+/g, "-");
    if (!loaded.includes(`family=${google}`) && !loaded.includes(`f[]=${share}`) && !loaded.includes("@font-face") && !loaded.includes("@fontsource"))
      fontFinding("this family is never loaded; add its Google Fonts or Fontshare link to index.html", family);
  }
}

// Pages that skip the kit invent their own layout, which is where pages stop matching each other.
const pages = files.filter((f) => /[\\/]pages[\\/]/.test(f));
for (const file of pages) {
  const text = readFileSync(file, "utf8");
  if (!/from ["']@\/kit["']|from ["']\.\.\/kit/.test(text))
    found.push({ file, line: 1, name: "page skips the kit", fix: "build pages from @/kit (Section, Grid, Heading, Card…)", code: "" });
}

if (!found.length) {
  console.log(`ui-check: ${files.length} files clean.`);
  process.exit(0);
}
const by = new Map();
for (const f of found) by.set(f.name, [...(by.get(f.name) ?? []), f]);
for (const [name, list] of by) {
  console.log(`\n${name} (${list.length}): ${list[0].fix}`);
  for (const f of list.slice(0, 12)) console.log(`  ${relative(root, f.file)}:${f.line}  ${f.code}`);
  if (list.length > 12) console.log(`  … ${list.length - 12} more`);
}
console.log(`\nui-check: ${found.length} findings in ${new Set(found.map((f) => f.file)).size} files.`);
process.exit(1);
